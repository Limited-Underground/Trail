cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET "esp32s3" CACHE STRING "Evaluation target")
set(PROJECT_VER "ot242-enrolled-completion-v1")
get_filename_component(OT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
set(OT_RADIOLIB "${OT_ROOT}/tests/benchmarks/crypto/esp_idf/noise_xk_ready_radio/managed_components/jgromes__radiolib")
set(EXTRA_COMPONENT_DIRS "${OT_RADIOLIB}" "${OT_ROOT}/tests/benchmarks/crypto/esp_idf/espressif_libsodium_1_0_22/managed_components/espressif__libsodium")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
set(CMAKE_C_COMPILER_LAUNCHER "${CMAKE_COMMAND};-E;env;SOURCE_DATE_EPOCH=0")
set(CMAKE_CXX_COMPILER_LAUNCHER "${CMAKE_COMMAND};-E;env;SOURCE_DATE_EPOCH=0")
idf_build_get_property(OT_PYTHON PYTHON)
execute_process(COMMAND "${OT_PYTHON}" "${OT_ROOT}/tools/security_evaluation_inputs.py" RESULT_VARIABLE OT_INPUTS)
if(NOT OT_INPUTS EQUAL 0)
    message(FATAL_ERROR "Evaluation source admission failed")
endif()
set(OT_CORE_SOURCE "${OT_RADIOLIB}/src/modules/SX126x/SX126x.cpp")
set(OT_CONFIG_SOURCE "${OT_RADIOLIB}/src/modules/SX126x/SX126x_config.cpp")
set(OT_RADIO_GENERATOR "${OT_ROOT}/tools/pair_radio_driver_source.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${OT_RADIO_GENERATOR}" "${OT_ROOT}/tools/radiolib_busy_source.py" "${OT_CORE_SOURCE}" "${OT_CONFIG_SOURCE}")
execute_process(COMMAND "${OT_PYTHON}" "${OT_RADIO_GENERATOR}"
    --core-source "${OT_CORE_SOURCE}" --config-source "${OT_CONFIG_SOURCE}"
    --core-output "${CMAKE_CURRENT_BINARY_DIR}/pair_SX126x.cpp"
    --config-output "${CMAKE_CURRENT_BINARY_DIR}/pair_SX126x_config.cpp"
    RESULT_VARIABLE OT_RADIO_GENERATED)
if(NOT OT_RADIO_GENERATED EQUAL 0)
    message(FATAL_ERROR "Pinned radio driver generation failed")
endif()
project(heltec_v4_enrolled_eval)
idf_component_get_property(OT_RADIO_TARGET jgromes__radiolib COMPONENT_LIB)
get_target_property(OT_RADIO_SOURCES ${OT_RADIO_TARGET} SOURCES)
set(OT_REPLACED_CORE 0)
set(OT_REPLACED_CONFIG 0)
set(OT_NEW_RADIO_SOURCES "")
foreach(OT_SOURCE IN LISTS OT_RADIO_SOURCES)
    if(OT_SOURCE STREQUAL OT_CORE_SOURCE)
        list(APPEND OT_NEW_RADIO_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/pair_SX126x.cpp")
        math(EXPR OT_REPLACED_CORE "${OT_REPLACED_CORE} + 1")
    elseif(OT_SOURCE STREQUAL OT_CONFIG_SOURCE)
        list(APPEND OT_NEW_RADIO_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/pair_SX126x_config.cpp")
        math(EXPR OT_REPLACED_CONFIG "${OT_REPLACED_CONFIG} + 1")
    else()
        list(APPEND OT_NEW_RADIO_SOURCES "${OT_SOURCE}")
    endif()
endforeach()
if(NOT OT_REPLACED_CORE EQUAL 1 OR NOT OT_REPLACED_CONFIG EQUAL 1)
    message(FATAL_ERROR "Expected exactly one of each pinned radio driver TU")
endif()
set_property(TARGET ${OT_RADIO_TARGET} PROPERTY SOURCES "${OT_NEW_RADIO_SOURCES}")
target_include_directories(${OT_RADIO_TARGET} PRIVATE "${OT_RADIOLIB}/src/modules/SX126x")
target_compile_options(${OT_RADIO_TARGET} PRIVATE "-ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=/PAIR_RADIO_BUILD_DIR")
