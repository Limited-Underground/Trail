cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET "esp32s3" CACHE STRING "Evaluation target")
set(PROJECT_VER "ot198-input-diag-v0")
get_filename_component(OT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
set(EXTRA_COMPONENT_DIRS "${OT_ROOT}/tests/benchmarks/crypto/esp_idf/espressif_libsodium_1_0_22/managed_components/espressif__libsodium")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
set(CMAKE_C_COMPILER_LAUNCHER "${CMAKE_COMMAND};-E;env;SOURCE_DATE_EPOCH=0")
set(CMAKE_CXX_COMPILER_LAUNCHER "${CMAKE_COMMAND};-E;env;SOURCE_DATE_EPOCH=0")
idf_build_get_property(OT_PYTHON PYTHON)
execute_process(COMMAND "${OT_PYTHON}" "${OT_ROOT}/tools/security_evaluation_inputs.py" RESULT_VARIABLE OT_INPUTS)
if(NOT OT_INPUTS EQUAL 0)
    message(FATAL_ERROR "Evaluation source admission failed")
endif()
project(heltec_v4_security_input_diag)

target_link_options(heltec_v4_security_input_diag.elf PRIVATE "-Wl,--wrap=_write_r" "-Wl,--wrap=esp_rom_output_tx_one_char" "-Wl,--wrap=ets_write_char_uart" "-Wl,--wrap=uart_tx_one_char")
